/// Incorrect forward copy of [0, 2) to [1, 3): the second read observes the
/// first write instead of the original source cell.
lemma negative_control(before : @arrays.SmtArray[@integer.Integer, Int]) where {
  proof_ensure: copied(
    before, before,
    @arrays.store(@arrays.store(before, @integer.one(), @arrays.select(before, @integer.zero())), @integer.from_int(2), @arrays.select(before, @integer.zero())),
    @integer.from_int(3), @integer.from_int(3), @integer.zero(), @integer.one(), @integer.from_int(2),
  ),
} {}
